auto_qol_hub_envoy_on = {
 potential = { from = { is_ai = no has_country_flag = auto_envoy_enabled } }
 allow = { from = { is_ai = no has_country_flag = auto_envoy_enabled } }
 effect = {
  custom_tooltip = "auto_qol_menu.toggle_envoy_tt"
  hidden_effect = { from = { remove_country_flag = auto_envoy_enabled } }
 }
}

auto_qol_hub_envoy_off = {
 potential = { from = { is_ai = no NOT = { has_country_flag = auto_envoy_enabled } } }
 allow = { from = { is_ai = no NOT = { has_country_flag = auto_envoy_enabled } } }
 effect = {
  custom_tooltip = "auto_qol_menu.toggle_envoy_tt"
  hidden_effect = { from = { set_country_flag = auto_envoy_enabled country_event = { id = auto_envoy.100 } } }
 }
}
