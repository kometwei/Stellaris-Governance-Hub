# Country scope; ROOT must be the owning player country.
# Persistent targets are qualified by country, never shared between players.
auto_qol_assimilation_apply = {
	if = {
		limit = {
			is_ai = no
			has_country_flag = auto_qol_assimilation_enabled
			exists = species
			exists = event_target:auto_qol_assimilation_source@root
			NOT = { is_same_species = event_target:auto_qol_assimilation_source@root }
		}
		species = { save_event_target_as = auto_qol_assimilation_destination }
		# Do leaders first. No cloning, name randomization or portrait effects.
		if = {
			limit = { has_country_flag = auto_qol_assimilation_leaders }
			every_owned_leader = {
				limit = { auto_qol_assimilation_eligible_leader = yes }
				change_species = event_target:auto_qol_assimilation_destination
			}
			every_pool_leader = {
				limit = { auto_qol_assimilation_eligible_leader = yes }
				change_species = event_target:auto_qol_assimilation_destination
			}
		}
		every_owned_pop_group = {
			limit = { is_same_species = event_target:auto_qol_assimilation_source@root }
			change_species = event_target:auto_qol_assimilation_destination
		}
	}
}

# Start a fresh browse session. The seen flag only controls pagination.
auto_qol_assimilation_reset_picker = {
	every_owned_species = {
		remove_species_flag = auto_qol_assimilation_seen@root
	}
}
