# 舰队数值加减计算器动态生效脚本

auto_qol_reapply_navy_cap = {
	# 移除旧的动态舰容加成
	remove_modifier = qol_custom_navy_cap_modifier
	
	# 如果变量大于 0，应用当前的自定义数值
	if = {
		limit = {
			check_variable = { which = qol_custom_navy_cap value > 0 }
		}
		add_modifier = {
			modifier = qol_custom_navy_cap_modifier
			days = -1
			multiplier = qol_custom_navy_cap
		}
	}
}

# 替换旧加成，清零时只移除本模组的自定义修正
auto_qol_reapply_starbase_module_speed = {
	remove_modifier = qol_custom_starbase_module_speed_modifier
	remove_modifier = qol_custom_starbase_upgrade_speed_modifier
	set_variable = { which = qol_custom_starbase_upgrade_speed value = 0 }
	if = {
		limit = { check_variable = { which = qol_custom_starbase_module_speed value > 0 } }
		add_modifier = {
			modifier = qol_custom_starbase_module_speed_modifier
			days = -1
			multiplier = qol_custom_starbase_module_speed
		}
	}
}

# 合并旧存档的基地建设/升级加成，只迁移一次，取两者较高值。
auto_qol_prepare_starbase_construction = {
	if = {
		limit = { NOT = { has_country_flag = auto_qol_starbase_construction_merged } }
		if = {
			limit = { NOT = { is_variable_set = qol_custom_starbase_module_speed } }
			set_variable = { which = qol_custom_starbase_module_speed value = 0 }
		}
		if = {
			limit = { NOT = { is_variable_set = qol_custom_starbase_upgrade_speed } }
			set_variable = { which = qol_custom_starbase_upgrade_speed value = 0 }
		}
		if = {
			limit = { check_variable = { which = qol_custom_starbase_upgrade_speed value > qol_custom_starbase_module_speed } }
			set_variable = { which = qol_custom_starbase_module_speed value = qol_custom_starbase_upgrade_speed }
		}
		set_country_flag = auto_qol_starbase_construction_merged
	}
	auto_qol_reapply_starbase_module_speed = yes
}
