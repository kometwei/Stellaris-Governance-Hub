namespace = auto_qol_council_pool

# Schedule one deferred trim when generation crosses the cap. Killing the new
# leader directly inside on_added_to_leader_pool can make the engine refill it
# repeatedly, so the actual trim runs after pool generation has settled.
country_event = {
	id = auto_qol_council_pool.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		has_country_flag = leader_pool_opt_enabled
		NOT = { has_country_flag = auto_qol_pool_trim_scheduled }
		count_pool_leader = {
			limit = { always = yes }
			count > 10
		}
	}

	immediate = {
		set_timed_country_flag = { flag = auto_qol_pool_trim_scheduled days = 2 }
		country_event = { id = auto_qol_council_pool.3 days = 1 }
	}
}

country_event = {
	id = auto_qol_council_pool.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		has_country_flag = leader_pool_opt_enabled
	}

	immediate = {
		auto_qol_trim_leader_pool_to_10 = yes
	}
}

		name = "auto_qol_council_pool.clean"
