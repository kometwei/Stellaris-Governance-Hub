# ========================================================
# 舰队容量与指挥上限控制台
# ========================================================
country_event = {
	id = auto_qol_fleet_menu.1
	title = "auto_qol_fleet_menu.1.name"
	desc = "auto_qol_fleet_menu.1.desc"
	picture = GFX_evt_fleet_neutral
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { is_variable_set = qol_custom_navy_cap } }
			set_variable = { which = qol_custom_navy_cap value = 0 }
		}
		if = {
			limit = { NOT = { is_variable_set = qol_custom_cmd_limit } }
			set_variable = { which = qol_custom_cmd_limit value = 0 }
		}
	}

	# 1. 进入舰队容量调节子菜单
	option = {
		name = "auto_qol_fleet_menu.go_navy_cap"
		custom_tooltip = "auto_qol_fleet_menu.go_navy_cap_tt"
		country_event = { id = auto_qol_navy_cap_menu.1 }
	}

	# 2. 进入指挥上限调节子菜单
	option = {
		name = "auto_qol_fleet_menu.go_cmd_limit"
		custom_tooltip = "auto_qol_fleet_menu.go_cmd_limit_tt"
		country_event = { id = auto_qol_cmd_limit_menu.1 }
	}

	# 3. 预设：无敌巨舰 (+10000 舰容, +500 指挥上限)
	option = {
		name = "auto_qol_fleet_menu.preset_super"
		custom_tooltip = "auto_qol_fleet_menu.preset_super_tt"
		change_variable = { which = qol_custom_navy_cap value = 10000 }
		change_variable = { which = qol_custom_cmd_limit value = 500 }
		auto_qol_reapply_navy_cap = yes
		auto_qol_reapply_cmd_limit = yes
		country_event = { id = auto_qol_fleet_menu.1 }
	}

	# 4. 清空所有额外舰容与指挥上限
	option = {
		name = "auto_qol_fleet_menu.clear_all"
		custom_tooltip = "auto_qol_fleet_menu.clear_all_tt"
		set_variable = { which = qol_custom_navy_cap value = 0 }
		set_variable = { which = qol_custom_cmd_limit value = 0 }
		auto_qol_reapply_navy_cap = yes
		auto_qol_reapply_cmd_limit = yes
		country_event = { id = auto_qol_fleet_menu.1 }
	}

	# 5. 返回助手主控制台
	option = {
		name = "auto_qol_fleet_menu.back"
		country_event = { id = auto_qol_menu.1 }
	}

	# 6. 离开并关闭界面 (支持右上角X与ESC)
	option = {
		name = "auto_qol.close_direct"
		default_hide_option = yes
	}
}
